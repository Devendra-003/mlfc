import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score, f1_score,
    confusion_matrix, classification_report
)
from statsmodels.stats.outliers_influence import variance_inflation_factor

df = pd.read_csv(r"C:\Users\AI18\Downloads\heart_disease_uci.csv")

print("First Five Records:")
print(df.head())

print("\nDataset Shape:")
print(df.shape)

print("\nColumn Name:")
print(df.columns)

print("\nDataset Information:")
print(df.info())

print("\nStatistical Summary:")
print(df.describe())

display(df.describe(include="all"))
print("\nMissing values: ")
print(df.isnull().sum()[df.isnull().sum() > 0])
print("Duplicate record: ", df.duplicated().sum())

if "num" not in df.columns:
    raise ValueError("Column 'num' was not found. Check the dataset.")

df["target"] = (df["num"] > 0).astype(int)
print(df["target"].value_counts())

sns.countplot(x="target", data=df)
plt.title("Heart Disease Class Distribution")
plt.xlabel("0 = No Disease, 1 = Disease")
plt.ylabel("Number of Patients")
plt.show()

print("\nFeature Extraction..")
X = df.drop(columns=["target", "num"])
y = df["target"]

numeric_features = X.select_dtypes(include=["int64", "float64"]).columns.tolist()
categorical_features = X.select_dtypes(include=["object", "category", "bool"]).columns.tolist()

print("Numerical Features: ", numeric_features)
print("Categorical Features: ", categorical_features)

print("\nSplitting Data into Training and testing sets: ")
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.20, random_state=42, stratify=y
)
print("Training Samples: ", X_train.shape[0])
print("Testing Samples: ", X_test.shape[0])

numeric_transformer = Pipeline([
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler())
])

categorical_transformer = Pipeline([
    ("imputer", SimpleImputer(strategy="most_frequent")),
    ("encoder", OneHotEncoder(handle_unknown="ignore", sparse_output=False))
])

preprocessor = ColumnTransformer([
    ("num", numeric_transformer, numeric_features),
    ("cat", categorical_transformer, categorical_features)
])

baseline_model = Pipeline([
    ("preprocessor", preprocessor),
    ("classifier", LogisticRegression(penalty=None, max_iter=5000))
])
baseline_model.fit(X_train, y_train)
y_train_base = baseline_model.predict(X_train)
y_test_base = baseline_model.predict(X_test)

l2_model = Pipeline([
    ("preprocessor", preprocessor),
    ("classifier", LogisticRegression(
        penalty="l2", C=1.0, solver="lbfgs", max_iter=5000
    ))
])
l2_model.fit(X_train, y_train)
y_train_l2 = l2_model.predict(X_train)
y_test_l2 = l2_model.predict(X_test)

l1_model = Pipeline([
    ("preprocessor", preprocessor),
    ("classifier", LogisticRegression(
        penalty="l1", C=1.0, solver="liblinear", max_iter=5000
    ))
])
l1_model.fit(X_train, y_train)
y_train_l1 = l1_model.predict(X_train)
y_test_l1 = l1_model.predict(X_test)

def evaluate(name, ytr, ptr, yte, pte):
    return {
        "Model": name,
        "Training Accuracy": accuracy_score(ytr, ptr),
        "Testing Accuracy": accuracy_score(yte, pte),
        "Precision": precision_score(yte, pte, zero_division=0),
        "Recall": recall_score(yte, pte, zero_division=0),
        "F1-Score": f1_score(yte, pte, zero_division=0)
    }

results = pd.DataFrame([
    evaluate("Logistic Regression", y_train, y_train_base, y_test, y_test_base),
    evaluate("L2 Regularization", y_train, y_train_l2, y_test, y_test_l2),
    evaluate("L1 Regularization", y_train, y_train_l1, y_test, y_test_l1)
])
display(results.round(4))

results.set_index("Model")[["Training Accuracy", "Testing Accuracy"]].plot(
    kind="bar", figsize=(9, 5)
)
plt.title("Training vs Testing accuracy")
plt.ylabel("Accuracy")
plt.ylim(0, 1)
plt.xticks(rotation=0)
plt.tight_layout()
plt.show()

plt.figure(figsize=(12, 8))
sns.heatmap(X[numeric_features].corr(), annot=True, cmap="coolwarm", fmt=".2f")
plt.title("Correlation Matrix of Numerical Features")
plt.tight_layout()
plt.show()

X_vif = X[numeric_features].copy().fillna(X[numeric_features].median())
vif = pd.DataFrame({
    "Feature": X_vif.columns,
    "VIF": [variance_inflation_factor(X_vif.values, i) for i in range(X_vif.shape[1])]
}).sort_values("VIF", ascending=False)
display(vif)

feature_names = l1_model.named_steps["preprocessor"].get_feature_names_out()
l2_coef = l2_model.named_steps["classifier"].coef_[0]
l1_coef = l1_model.named_steps["classifier"].coef_[0]

coef_comparison = pd.DataFrame({
    "Feature": feature_names,
    "L2_Coefficient": l2_coef,
    "L1_Coefficient": l1_coef
})
display(coef_comparison)

selected = coef_comparison[coef_comparison["L1_Coefficient"] != 0]
zeroed = coef_comparison[coef_comparison["L1_Coefficient"] == 0]

print("Total processed features:", len(coef_comparison))
print("Selected by L1:", len(selected))
print("Set to zero by L1:", len(zeroed))
print("\nFeatures selected by L1:")
display(selected)

coef_comparison.set_index("Feature").plot(kind="bar", figsize=(14, 6))
plt.title("L1 vs L2 Logistic Regression Coefficients")
plt.ylabel("Coefficient")
plt.xticks(rotation=75)
plt.tight_layout()
plt.show()

C_values = [0.01, 0.1, 1, 10, 100]
l1_strength, l2_strength = [], []

for C in C_values:
    l1 = Pipeline([
        ("preprocessor", preprocessor),
        ("classifier", LogisticRegression(
            penalty="l1", C=C, solver="liblinear", max_iter=5000
        ))
    ])
    l1.fit(X_train, y_train)
    p1 = l1.predict(X_test)

    l2 = Pipeline([
        ("preprocessor", preprocessor),
        ("classifier", LogisticRegression(
            penalty="l2", C=C, solver="lbfgs", max_iter=5000
        ))
    ])
    l2.fit(X_train, y_train)
    p2 = l2.predict(X_test)

    l1_strength.append({
        "C": C,
        "Test Accuracy": accuracy_score(y_test, p1),
        "F1": f1_score(y_test, p1, zero_division=0),
        "Non-zero features": np.count_nonzero(l1.named_steps["classifier"].coef_[0])
    })
    l2_strength.append({
        "C": C,
        "Test Accuracy": accuracy_score(y_test, p2),
        "F1": f1_score(y_test, p2, zero_division=0)
    })

l1_strength = pd.DataFrame(l1_strength)
l2_strength = pd.DataFrame(l2_strength)

print("L1 Results")
display(l1_strength.round(4))
print("L2 Results")
display(l2_strength.round(4))

print("L2 Regularization - Classification Report")
print(classification_report(y_test, y_test_l2, zero_division=0))

print("L1 Regularization - Classification Report")
print(classification_report(y_test, y_test_l1, zero_division=0))

fig, axes = plt.subplots(1, 2, figsize=(11, 4))
sns.heatmap(confusion_matrix(y_test, y_test_l2), annot=True, fmt="d", cmap="Blues", ax=axes[0])
axes[0].set_title("L2 Regularization")
axes[0].set_xlabel("Predicted")
axes[0].set_ylabel("Actual")

sns.heatmap(confusion_matrix(y_test, y_test_l1), annot=True, fmt="d", cmap="Greens", ax=axes[1])
axes[1].set_title("L1 Regularization")
axes[1].set_xlabel("Predicted")
axes[1].set_ylabel("Actual")
plt.tight_layout()
plt.show()
